import RPi.GPIO as GPIO
import time

GPIO.setmode(GPIO.BCM)

# === Sensor Pin Reference ===
# DHT11 / DHT22 (Temp & Humidity) - GPIO 4 (use with Adafruit_DHT library)
# Touch Sensor (TTP223)           - GPIO 17
# IR Tracking Sensor              - GPIO 18
# Mercury Tilt Switch             - GPIO 23
# Laser Emitter (Output)          - GPIO 24
# Button                          - GPIO 25
# IR Emitter (Output)             - GPIO 26
# Ball Switch                     - GPIO 19
# Tap Sensor (Vibration/SW-420)   - GPIO 16

SENSOR_PIN = 17  # 👈 Change this to match your sensor pin (e.g., 23 for tilt switch)

GPIO.setup(SENSOR_PIN, GPIO.IN)

try:
    while True:
        if GPIO.input(SENSOR_PIN):
            print("Sensor Triggered")  # 👈 Change this to your desired output
        time.sleep(0.5)

except KeyboardInterrupt:
    GPIO.cleanup()
